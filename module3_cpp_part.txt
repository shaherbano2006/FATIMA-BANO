// [Module 3] Sets the details every person has. Nobody is the culprit yet.
Character::Character(string name, string role, string nation)
{
    charactername = name;
    characterrole = role;
    nationality = nation;
    isculprit = 0;
    interviewed = 0;
}

// [Module 3 + 9] Virtual destructor: deleting a Character* also runs the
// destructor of the real child (Cook, PA, ...).
Character::~Character()
{
}

// [Module 3] Prints the top part of a character card.
void Character::showBasicInfo() const
{
    printLine('-');
    cout << "  |  Name        : " << charactername << endl;
    cout << "  |  Role        : " << characterrole << endl;
    cout << "  |  Nationality : " << nationality << endl;
}

// [Module 3] Getter: name.
string Character::getName() const
{
    return charactername;
}

string Character::getRole() const

    return characterrole;
}

// [Module 3] Getter: nationality.
string Character::getNationality() const

    return nationality;
}

// [Module 3] Getter: 1 if this person is the culprit, otherwise 0.
int Character::getIsCulprit() const

}

// [Module 3] Getter: 1 if already interviewed, otherwise 0.
int Character::isInterviewed() const
{
    return interviewed;
}

// [Module 3] Setter: name.
void Character::setName(string name)
{
    charactername = name;
}

// [Module 3] Setter: role.
void Character::setRole(string role)
{
    characterrole = role;
}

// [Module 3] Setter: nationality.
void Character::setNationality(string nation)
{
    nationality = nation;
}

// [Module 3] Setter: interviewed (1 or 0).
void Character::setInterviewed(int value)
{
    interviewed = value;
}

// [Module 3 + 9] 1 if the detective's credibility is high enough to talk.
// getRequiredCredibility() is virtual, so every role uses its own limit.
int Character::canBeInterviewed(int credibility) const
{
    int needed = getRequiredCredibility();
    int answer = 0;
    if (credibility >= needed)
    {
        answer = 1;
    }
    return answer;
}

// [Module 11 - Operator Overloading + Friend] cout << person
// Prints e.g.  Mr.Toby (Servant)
ostream& operator<<(ostream& out, const Character& person)
{
    out << person.charactername << " (" << person.characterrole << ")";
    return out;
}

// [Module 11 - Operator Overloading + Friend] cin >> person
// Reads name, role and nationality (no spaces, use _).
istream& operator>>(istream& in, Character& person)
{
    in >> person.charactername >> person.characterrole >> person.nationality;
    return in;
}

// ############################################################################
//  MODULE 3 : CLASS Detective  (Composition with TileRack - Module 10)
// ############################################################################

// [Module 3] The player: 100 credibility, 0 turns, a 7-slot and an 8-slot rack.
Detective::Detective() : Character("Detective", "Detective", "Malay"),
                         rackone("Word 1 Rack", 7), racktwo("Word 2 Rack", 8)
{
    credibility = 100;
    turns = 0;
}

// [Module 3] Copy constructor (deep copy of both racks).
Detective::Detective(const Detective& other) : Character(other),
                                               rackone(other.rackone), racktwo(other.racktwo)
{
    credibility = other.credibility;
    turns = other.turns;
}

// [Module 3 + 11] Assignment operator (deep copy).
Detective& Detective::operator=(const Detective& other)
{
    if (this != &other)
    {
        Character::operator=(other);        // copy the Character part first
        credibility = other.credibility;
        turns = other.turns;
        rackone = other.rackone;
        racktwo = other.racktwo;
    }
    return *this;
}

// [Module 3] Destructor (the racks clean themselves).
Detective::~Detective()
{
}

// [Module 2] Gets the detective ready for a new case.
void Detective::resetForNewGame(string name, int startcredibility)
{
    charactername = name;
    credibility = startcredibility;
    turns = 0;
    rackone.clear();
    racktwo.clear();
}

// [Module 3] Getter: credibility.
int Detective::getCredibility() const
{
    return credibility;
}

// [Module 3] Setter: credibility (never below 0).
void Detective::setCredibility(int value)
{
    if (value < 0)
    {
        value = 0;
    }
    credibility = value;
}

// [Module 3] Cuts credibility after a mistake (stops at 0).
void Detective::reduceCredibility(int amount)
{
    int newvalue = credibility - amount;
    setCredibility(newvalue);
}

// [Module 3] Getter: turns used. (The GameEngine counts, the detective only stores.)
int Detective::getTurns() const
{
    return turns;
}

// [Module 3] Setter: turns used (never below 0).
void Detective::setTurns(int value)
{
    if (value < 0)
    {
        value = 0;
    }
    turns = value;
}

// [Module 3 + 5] Gives the REAL Word 1 rack (the & means "not a copy"),
// so letters added to it stay inside the detective.
TileRack& Detective::getRackOne()
{
    return rackone;
}

// [Module 3 + 5] Gives the REAL Word 2 rack.
TileRack& Detective::getRackTwo()
{
    return racktwo;
}

// [Module 5] Prints both racks.
void Detective::showRacks() const
{
    rackone.display();
    racktwo.display();
}

// [Module 9] Detective's own display.
void Detective::display() const
{
    showBasicInfo();
    cout << "  |  Credibility : " << credibility << endl;
    cout << "  |  Turns used  : " << turns << endl;
    printLine('-');
    showRacks();
}

// [Module 3 + 7] The detective collects clues, he never gives one.
Clue Detective::giveClue() const
{
    throw InvalidInputException("The detective collects clues, he does not give them.");
}

// [Module 3] The detective has no letter, so this does nothing.
void Detective::setClue(char letter, int position)
{
}

// [Module 3] No credibility needed.
int Detective::getRequiredCredibility() const
{
    return 0;
}

// [Module 3] The detective is not a suspect.
int Detective::isSuspect() const
{
    return 0;
}

// [Module 9] The detective's own line.
string Detective::speak() const
{
    return "I ask the questions around here.";
}
